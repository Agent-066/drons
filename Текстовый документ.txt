class YOLOONNXRecognizer:
    """YOLO распознаватель без PyTorch, используя ONNX Runtime"""
    
    def __init__(self, onnx_path, confidence_threshold=0.5, nms_threshold=0.45):
        """
        Инициализация с ONNX моделью
        
        Args:
            onnx_path: путь к .onnx файлу
            confidence_threshold: порог уверенности
            nms_threshold: порог NMS
        """
        # Проверка наличия onnxruntime
        try:
            import onnxruntime as ort
        except ImportError:
            raise ImportError("Установите onnxruntime: pip install onnxruntime")
        
        # Определение провайдера (CPU/GPU)
        available_providers = ort.get_available_providers()
        print(f"📊 Доступные провайдеры: {available_providers}")
        
        # Выбор лучшего провайдера
        if 'CUDAExecutionProvider' in available_providers:
            self.provider = 'CUDAExecutionProvider'
            print("✅ Используется NVIDIA GPU")
        elif 'ROCMExecutionProvider' in available_providers:
            self.provider = 'ROCMExecutionProvider'
            print("✅ Используется AMD GPU (ROCm)")
        elif 'OpenVINOExecutionProvider' in available_providers:
            self.provider = 'OpenVINOExecutionProvider'
            print("✅ Используется Intel OpenVINO")
        else:
            self.provider = 'CPUExecutionProvider'
            print("⚠️ Используется CPU")
        
        # Загрузка модели
        self.session = ort.InferenceSession(onnx_path, providers=[self.provider])
        
        # Параметры
        self.confidence_threshold = confidence_threshold
        self.nms_threshold = nms_threshold
        
        # Получение информации о входах/выходах
        self.input_name = self.session.get_inputs()[0].name
        self.output_names = [output.name for output in self.session.get_outputs()]
        
        # Определение размера входа
        input_shape = self.session.get_inputs()[0].shape
        self.input_size = (input_shape[2], input_shape[3])  # height, width
        
        # Цвета для классов
        self.colors = [
            (0, 255, 0), (255, 0, 0), (0, 0, 255),
            (255, 255, 0), (255, 0, 255), (0, 255, 255)
        ]
        
        # Классы (если есть)
        self.class_names = self.load_class_names()
        
        # FPS счетчик
        self.fps = 0
        self.frame_count = 0
        self.last_time = time.time()
        
        print(f"✅ Модель загружена, размер входа: {self.input_size}")
    
    def load_class_names(self, classes_path="coco.names"):
        """Загрузка имен классов"""
        # Стандартные классы COCO
        default_classes = [
            'person', 'bicycle', 'car', 'motorcycle', 'airplane', 'bus', 'train', 'truck',
            'boat', 'traffic light', 'fire hydrant', 'stop sign', 'parking meter', 'bench',
            'bird', 'cat', 'dog', 'horse', 'sheep', 'cow', 'elephant', 'bear', 'zebra',
            'giraffe', 'backpack', 'umbrella', 'handbag', 'tie', 'suitcase', 'frisbee',
            'skis', 'snowboard', 'sports ball', 'kite', 'baseball bat', 'baseball glove',
            'skateboard', 'surfboard', 'tennis racket', 'bottle', 'wine glass', 'cup',
            'fork', 'knife', 'spoon', 'bowl', 'banana', 'apple', 'sandwich', 'orange',
            'broccoli', 'carrot', 'hot dog', 'pizza', 'donut', 'cake', 'chair', 'couch',
            'potted plant', 'bed', 'dining table', 'toilet', 'tv', 'laptop', 'mouse',
            'remote', 'keyboard', 'cell phone', 'microwave', 'oven', 'toaster', 'sink',
            'refrigerator', 'book', 'clock', 'vase', 'scissors', 'teddy bear', 'hair drier',
            'toothbrush'
        ]
        
        if os.path.exists(classes_path):
            with open(classes_path, 'r') as f:
                return [line.strip() for line in f.readlines()]
        return default_classes
    
    def preprocess(self, image):
        """Предобработка изображения для ONNX"""
        # Изменение размера
        resized = cv2.resize(image, self.input_size)
        
        # Нормализация
        input_data = resized.astype(np.float32) / 255.0
        
        # Изменение формата (HWC -> CHW)
        input_data = np.transpose(input_data, (2, 0, 1))
        
        # Добавление batch dimension
        input_data = np.expand_dims(input_data, axis=0)
        
        return input_data
    
    def postprocess(self, outputs, original_shape):
        """Постобработка результатов"""
        detections = []
        
        # Обработка в зависимости от формата вывода
        if len(outputs) == 1:
            # Формат: [batch, num_detections, 6] (x1,y1,x2,y2,conf,class)
            output = outputs[0]
            for detection in output[0]:
                if len(detection) >= 6:
                    x1, y1, x2, y2, confidence, class_id = detection[:6]
                    
                    if confidence > self.confidence_threshold:
                        # Масштабирование координат
                        h, w = original_shape[:2]
                        x1 = int(x1 * w)
                        y1 = int(y1 * h)
                        x2 = int(x2 * w)
                        y2 = int(y2 * h)
                        
                        detections.append({
                            'bbox': [x1, y1, x2, y2],
                            'confidence': float(confidence),
                            'class_id': int(class_id),
                            'class_name': self.class_names[int(class_id)] if int(class_id) < len(self.class_names) else f"class_{int(class_id)}"
                        })
        
        return detections
    
    def process_frame(self, frame, show_boxes=True, show_labels=True):
        """Обработка одного кадра"""
        # Сохраняем оригинальный размер
        original_shape = frame.shape
        
        # Предобработка
        input_data = self.preprocess(frame)
        
        # Инференс
        outputs = self.session.run(self.output_names, {self.input_name: input_data})
        
        # Постобработка
        detected_objects = self.postprocess(outputs, original_shape)
        
        # Рисование результатов
        annotated_frame = frame.copy()
        
        if show_boxes:
            for obj in detected_objects:
                x1, y1, x2, y2 = obj['bbox']
                color = self.colors[obj['class_id'] % len(self.colors)]
                
                # Рисуем рамку
                cv2.rectangle(annotated_frame, (x1, y1), (x2, y2), color, 2)
                
                if show_labels:
                    label = f"{obj['class_name']}: {obj['confidence']:.2f}"
                    label_size = cv2.getTextSize(label, cv2.FONT_HERSHEY_SIMPLEX, 0.5, 2)[0]
                    cv2.rectangle(annotated_frame, (x1, y1 - label_size[1] - 10),
                                (x1 + label_size[0], y1), color, -1)
                    cv2.putText(annotated_frame, label, (x1, y1 - 5),
                              cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 2)
        
        # Обновление FPS
        self.update_fps()
        self.add_frame_info(annotated_frame, detected_objects)
        
        return annotated_frame, detected_objects
    
    def update_fps(self):
        """Обновление FPS"""
        self.frame_count += 1
        current_time = time.time()
        if current_time - self.last_time >= 1.0:
            self.fps = self.frame_count
            self.frame_count = 0
            self.last_time = current_time
    
    def add_frame_info(self, frame, detected_objects):
        """Добавление информации на кадр"""
        cv2.putText(frame, f"FPS: {self.fps}", (10, 30),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        cv2.putText(frame, f"Objects: {len(detected_objects)}", (10, 60),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        cv2.putText(frame, f"Backend: ONNX-{self.provider.replace('ExecutionProvider', '')}", (10, 90),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 0), 1)


### 2️⃣ **Еще проще - OpenCV DNN (без дополнительных библиотек)**

```python
class YOLOOpenCVDNNRecognizer:
    """YOLO распознаватель используя только OpenCV"""
    
    def __init__(self, weights_path, config_path, classes_path="coco.names"):
        """
        Args:
            weights_path: путь к .weights файлу (Darknet)
            config_path: путь к .cfg файлу
            classes_path: путь к файлу с классами
        """
        # Загрузка сети
        self.net = cv2.dnn.readNet(weights_path, config_path)
        
        # Использование CUDA если доступно
        if cv2.cuda.getCudaEnabledDeviceCount() > 0:
            print("✅ Используется OpenCV CUDA")
            self.net.setPreferableBackend(cv2.dnn.DNN_BACKEND_CUDA)
            self.net.setPreferableTarget(cv2.dnn.DNN_TARGET_CUDA)
        else:
            print("⚠️ Используется CPU")
            self.net.setPreferableBackend(cv2.dnn.DNN_BACKEND_OPENCV)
            self.net.setPreferableTarget(cv2.dnn.DNN_TARGET_CPU)
        
        # Загрузка классов
        self.class_names = []
        if os.path.exists(classes_path):
            with open(classes_path, 'r') as f:
                self.class_names = [line.strip() for line in f.readlines()]
        
        # Параметры
        self.confidence_threshold = 0.5
        self.nms_threshold = 0.4
        
        # Получение выходных слоев
        self.output_layers = self.net.getUnconnectedOutLayersNames()
        
        # Цвета
        self.colors = np.random.uniform(0, 255, size=(len(self.class_names), 3))
        
        # FPS счетчик
        self.fps = 0
        self.frame_count = 0
        self.last_time = time.time()
    
    def process_frame(self, frame):
        """Обработка кадра"""
        height, width = frame.shape[:2]
        
        # Создание блоба
        blob = cv2.dnn.blobFromImage(frame, 1/255.0, (416, 416), swapRB=True, crop=False)
        
        # Инференс
        self.net.setInput(blob)
        outputs = self.net.forward(self.output_layers)
        
        # Постобработка
        boxes, confidences, class_ids = [], [], []
        
        for output in outputs:
            for detection in output:
                scores = detection[5:]
                class_id = np.argmax(scores)
                confidence = scores[class_id]
                
                if confidence > self.confidence_threshold:
                    center_x = int(detection[0] * width)
                    center_y = int(detection[1] * height)
                    w = int(detection[2] * width)
                    h = int(detection[3] * height)
                    
                    x = int(center_x - w / 2)
                    y = int(center_y - h / 2)
                    
                    boxes.append([x, y, w, h])
                    confidences.append(float(confidence))
                    class_ids.append(class_id)
        
        # NMS
        indexes = cv2.dnn.NMSBoxes(boxes, confidences, self.confidence_threshold, self.nms_threshold)
        
        # Рисование результатов
        detected_objects = []
        if len(indexes) > 0:
            for i in indexes.flatten():
                x, y, w, h = boxes[i]
                color = self.colors[class_ids[i]]
                
                cv2.rectangle(frame, (x, y), (x + w, y + h), color, 2)
                label = f"{self.class_names[class_ids[i]]}: {confidences[i]:.2f}"
                cv2.putText(frame, label, (x, y - 5), cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)
                
                detected_objects.append({
                    'class_name': self.class_names[class_ids[i]],
                    'confidence': confidences[i],
                    'bbox': [x, y, x + w, y + h]
                })
        
        # Обновление FPS
        self.update_fps()
        cv2.putText(frame, f"FPS: {self.fps}", (10, 30),
                   cv2.FONT_HERSHEY_SIMPLEX, 0.7, (0, 255, 0), 2)
        
        return frame, detected_objects
    
    def update_fps(self):
        self.frame_count += 1
        if time.time() - self.last_time >= 1.0:
            self.fps = self.frame_count
            self.frame_count = 0
            self.last_time = time.time()
